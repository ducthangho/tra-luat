+# tra-luat executor/reviewer environment
+
+Both update scripts use `ops/cong_cu.yaml` and `ops/cong_cu.py` for the shared baseline: git, Python, PyYAML, jsonschema, actionlint, shellcheck, ripgrep and Bash. Linux system mode and Windows probe each baseline entry and invoke `ops/cai_cong_cu.sh` only for entries that are not `DAT`, with ShellCheck prepared before actionlint. Linux system mode bootstraps checker PyYAML only when its import is missing, handles nonzero probe exits without aborting repair under `set -e`, and requires each repaired tool to report `DAT` in the parent process. Linux actionlint installation uses child-scoped `AL_DICH=$HOME/.local/bin`; that directory is already on the parent's PATH, so it does not depend on the installer's temporary default or child PATH changes. Linux `--user-only` installs pinned Python dependencies in an isolated venv and runs the same checker. `actionlint` is intentionally pinned to 1.7.7 by the repository manifest; the canary checks its ShellCheck integration. `--latest`/`-Latest` upgrades only the tools named in `--tools`/`-Tools` (default `claude,codex,agy,nlm,gws`); a tool already at the target version is not reinstalled. System tools use the configured OS package manager, so "latest" for those means the newest version in that repository, not necessarily the upstream release. Python and Node major updates can affect existing environments and are never selected by default.
+
+Run the scripts without `--apply`/`-Apply` to inspect. Check mode installs nothing and exits with the checker's exit code. The apply mode refuses to proceed while related tool/runtime processes are active (Claude, Codex, AGY, nlm, gws, uv, Node, Python; on Windows also WSL); on Linux it also refuses when `pgrep` is unavailable. An unknown `--tools` entry is rejected before any installation. `--persist-path`/`-PersistPath` is opt-in and adds user CLI directories idempotently: on Windows it edits the raw `HKCU\Environment\Path` value so existing `%VAR%` entries are preserved; on Linux it appends one marked, POSIX-compatible block to `~/.profile` and `~/.bashrc` that also sets `NPM_CONFIG_PREFIX=$HOME/.local` (do not use it together with nvm). Reopen terminals after changing persistent PATH.
+
+The updater reports only whether `TL_MASTER_KEY`, `TL_SECRETS_FILEID`, `TL_AGENT_CREDS_FILEID`, `TL_WEBAPP`, `TL_TOKEN` and `TL_PROFILE` have nonempty values. It never prints values, generates keys, copies credentials between Windows and WSL, runs `secrets-stage`, or runs `tl bootstrap`. The tra-luat config load order is environment `TL_WEBAPP`/`TL_TOKEN`, explicit `--config`, `~/.tra_luat_config.json`, then `/tmp/tra_luat_config.json`. WSL has its own HOME and must be bootstrapped separately with the repository's documented secret flow once the operator has securely supplied its key.

Supplemental CLIs with a successful nonempty version probe are retained; baseline entries already `DAT` are retained. Missing tools are installed regardless of `-Tools`; that list selects upgrades only. Chocolatey operations require an already elevated PowerShell and stop with an explicit prerequisite error otherwise; the updater neither installs a package manager nor opens an elevation prompt. Windows persistent PATH remains opt-in with `-PersistPath`.

+function Add-UserPath([string]$directory) {
+    if (-not $directory -or -not (Test-Path -LiteralPath $directory -PathType Container)) { return }
+    # Read and write HKCU\Environment\Path raw: [Environment]::Get/SetEnvironmentVariable expands
+    # %VAR% entries and rewrites the value as REG_SZ, silently destroying the user's references.
+    $key = [Microsoft.Win32.Registry]::CurrentUser.OpenSubKey('Environment', $true)
+    if (-not $key) { throw 'Cannot open HKCU\Environment for writing' }
+    try {
+        $raw = [string]$key.GetValue('Path', '', [Microsoft.Win32.RegistryValueOptions]::DoNotExpandEnvironmentNames)
+        if (-not (Test-PathListContains $raw $directory)) {
+            $kind = [Microsoft.Win32.RegistryValueKind]::ExpandString
+            if (($key.GetValueNames() -contains 'Path') -and
+                $key.GetValueKind('Path') -eq [Microsoft.Win32.RegistryValueKind]::String) {
+                $kind = [Microsoft.Win32.RegistryValueKind]::String
+            }
+            $items = @($raw -split ';' | Where-Object { $_ }) + $directory
+            $key.SetValue('Path', ($items -join ';'), $kind)
+            Write-Host "Added to user PATH: $directory"
+        }
+    } finally { $key.Dispose() }
+    if (-not (Test-PathListContains $env:Path $directory)) { $env:Path = "$env:Path;$directory" }
+}

+function Run([string]$name, [string[]]$arguments) {
+    $cmd = Get-Command $name -ErrorAction Stop
+    & $cmd.Source @arguments
+    if ($LASTEXITCODE -ne 0) { throw "$name exited $LASTEXITCODE" }
+}
+function Version([string]$name, [string[]]$arguments) {
+    $cmd = Get-Command $name -ErrorAction SilentlyContinue
+    if (-not $cmd) { return '(missing)' }
+    try {
+        $lines = @(& $cmd.Source @arguments 2>$null)
+        $versionLine = $lines | Where-Object { $_ -match '^version:\s*\d' } | Select-Object -First 1
+        if ($versionLine) { return ([string]$versionLine).Trim() }
+        return ([string]($lines | Select-Object -First 1)).Trim()
+    }
+    catch { return '(error)' }
+}

+    $arg = if ($name -eq 'actionlint') { @('-version') } else { @('--version') }
+    '{0,-12} {1}' -f $name, (Version $name $arg)

+    $_.ProcessName -match '^(claude|codex|codex-command-runner.*|agy|nlm|gws|uv|node|python|pythonw|py|wsl|wslhost|vmmemWSL)$'

+    # Plan missing system prerequisites before any package mutation. Never self-elevate.
+    $missingNode = -not (Test-ToolReady 'node')
+    $missingNpm = -not (Test-ToolReady 'npm')
+    if (-not $missingNode -and $missingNpm) {
+        throw 'Node is present but npm is missing or broken. Repair that Node installation first; refusing to replace an existing Node major.'

+        $json = @(& python $checker kiem-tra --chi $baseline --json)

+        $probe = ($json -join [Environment]::NewLine) | ConvertFrom-Json
+        $entries = @($probe.cong_cu)
+        if ($entries.Count -ne 1 -or $entries[0].ten -cne $baseline) { throw "Invalid baseline probe: $baseline" }
+        if ($probeExit -eq 0 -and $entries[0].trang_thai -ceq 'DAT') { continue }

+            if ($baseline -ceq 'actionlint') {

+                if ($LASTEXITCODE -ne 0 -or $translated.Count -ne 1 -or
+                    [string]::IsNullOrWhiteSpace([string]$translated[0])) {
+                    throw 'Could not translate actionlint user bin for Git Bash'
+                }
+                $env:AL_DICH = ([string]$translated[0]).Trim()
+            }
+            & $gitBash $unixInstaller $baseline

+        Sync-ToolPath $baseline
+        # Measure in the parent, which does not inherit the Git Bash installer's PATH.
+        $json = @(& python $checker kiem-tra --chi $baseline --json)

+        $probe = ($json -join [Environment]::NewLine) | ConvertFrom-Json
