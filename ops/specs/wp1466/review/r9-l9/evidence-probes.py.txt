from pathlib import Path
import subprocess,sys,json
O=Path(__file__).resolve().parent; C=O/'candidate'; root=O/'probe-root';(root/'docs').mkdir(parents=True,exist_ok=True)
p=root/'docs/PROTOCOL_DA_PHIEN.md';base=(C/'docs/PROTOCOL_DA_PHIEN.md').read_bytes()
keys=json.loads((O/'keys.json').read_text()); records=[]
def run(name,raw,rc,reason=''):
 p.write_bytes(raw)
 cmd=[sys.executable,'-B',str(C/'.claude/dev/packet_loop.py'),'doctrine-scan','--root',str(root)]
 r=subprocess.run(cmd,capture_output=True,text=True); msg=r.stdout+r.stderr
 ok=r.returncode==rc and reason in msg
 records.append(dict(name=name,cmd=cmd,rc=r.returncode,stdout=r.stdout,stderr=r.stderr,expected=rc,ok=ok))
 print(name,'RC',r.returncode,'PASS',ok,'REASON',msg.strip().replace('\n',' / ')[:250],flush=True)
 assert ok,name
run('base',base,0,'sạch')
for k,v in keys.items():
 if not v['order']:continue
 unit=(O/(k+'.raw')).read_bytes()
 for label,front in [('first',unit+b'\n\n'),('adjacent-heading',unit+b'\n'),('later-block',b'Ghi chu thu nghiem.\n\n'+unit+b'\n'),('leading-blanks',b'\n\n'+unit+b'\n\n'),('two-copies',unit+b'\n\n'+unit+b'\n\n')]:
  run(k+'-'+label,front+base,9,k+' : có '+('3' if label=='two-copies' else '2')+' bản')
 # Hai kiểu che được hợp đồng loại khỏi phép đếm; không nội dung thực thi.
 for label,front in [('fence',b'```text\n\n'+unit+b'\n\n```\n\n'),('comment',b'<!--\n\n'+unit+b'\n\n-->\n\n')]:
  run(k+'-'+label,front+base,0,'sạch')
 # Giới hạn dãy con so với đơn vị: phải ghi rõ, không tự đổi hợp đồng.
 for label,front in [('attached-before',b'Ghi chu thu nghiem.\n'+unit+b'\n\n'),('attached-after',unit+b'\nGhi chu thu nghiem.\n\n'),('space-line',b'Ghi chu thu nghiem.\n \n'+unit+b'\n\n')]:
  run(k+'-'+label,front+base,0,'sạch')
for k in ['K01','K03']:
 unit=(O/(k+'.raw')).read_bytes();run(k+'-whole-section-before',unit+b'\n\n'+base,9)
for label,front in [('plain',b'Ghi chu thu nghiem.\n\n'),('blanks',b'\n\n\n'),('comment-heading',b'<!--\n# Chu thich\n-->\n\n'),('fence-heading',b'```text\n# Vi du\n```\n\n')]:run('neutral-'+label,front+base,0,'sạch')
# Dời đơn vị khỏi địa chỉ sang phần đầu không được phép đếm (3) cứu phép (2).
for k in ['K02','K04','K05','K06','K07']:
 unit=(O/(k+'.raw')).read_bytes();assert base.count(unit)==1
 run(k+'-move-before',unit+b'\n\n'+base.replace(unit,b'Ghi chu thu nghiem.',1),9,'không còn nguyên')
# Không tiêu đề: thử đường lỗi của parser; không đường dẫn hoặc mã đối kháng.
nohead=b'\n'.join(b'Ghi chu '+l if l.startswith(b'#') else l for l in base.split(b'\n'))
run('no-headings',nohead,9,'mục chứa nó phải có đúng 1 lần, thấy 0')
(O/'probes.json').write_text(json.dumps(records,ensure_ascii=False,indent=2)); print('PROBES',len(records),'PASS',sum(r['ok'] for r in records))
